cmake_minimum_required(VERSION 3.21)
project(ElectraDesktop VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.5.3 REQUIRED COMPONENTS Widgets Network WebSockets Sql Svg Charts Test WebEngineWidgets WebChannel)
if(MSVC)
  add_compile_options(/utf-8 /W4)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(electra_common STATIC src/client.cpp src/client.h src/ui.cpp src/ui.h src/dialogs.cpp src/charger_picker.cpp src/avatar_editor.cpp src/station_map.cpp src/station_map.h src/visuals.cpp src/visuals.h resources.qrc)
target_include_directories(electra_common PUBLIC src)
target_link_libraries(electra_common PUBLIC Qt6::Widgets Qt6::Network Qt6::WebSockets Qt6::Sql Qt6::Svg Qt6::WebEngineWidgets Qt6::WebChannel)
qt_add_executable(electra-user src/user_main.cpp src/user_window.cpp src/user_window.h)
target_link_libraries(electra-user PRIVATE electra_common)
qt_add_executable(electra-admin src/admin_main.cpp src/admin_window.cpp src/admin_window.h)
target_link_libraries(electra-admin PRIVATE electra_common Qt6::Charts)
include(CTest)
if(BUILD_TESTING)
  qt_add_executable(test-editing tests/test_editing.cpp)
  target_link_libraries(test-editing PRIVATE electra_common Qt6::Test)
  add_test(NAME editing-interactions COMMAND test-editing -platform offscreen)
  qt_add_executable(test-dialogs tests/test_dialogs.cpp)
  target_link_libraries(test-dialogs PRIVATE electra_common Qt6::Test)
  add_test(NAME dialog-fit COMMAND test-dialogs -platform offscreen)
  qt_add_executable(test-visuals tests/test_visuals.cpp src/admin_window.cpp src/admin_window.h src/user_window.cpp src/user_window.h)
  target_link_libraries(test-visuals PRIVATE electra_common Qt6::Charts Qt6::Test)
  target_compile_definitions(test-visuals PRIVATE ELECTRA_FIXTURES_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  add_test(NAME native-pages COMMAND test-visuals -platform offscreen)
  qt_add_executable(test-map tests/test_map.cpp)
  target_link_libraries(test-map PRIVATE electra_common Qt6::Test)
  add_test(NAME map-interaction COMMAND test-map -platform offscreen)
  qt_add_executable(test-client tests/test_client.cpp)
  target_link_libraries(test-client PRIVATE electra_common Qt6::Test)
  add_test(NAME client-contract COMMAND test-client -platform offscreen)
  qt_add_executable(test-desktop tests/test_desktop.cpp src/admin_window.cpp src/admin_window.h src/user_window.cpp src/user_window.h)
  target_link_libraries(test-desktop PRIVATE electra_common Qt6::Charts Qt6::Test)
endif()
install(TARGETS electra-user electra-admin RUNTIME DESTINATION bin)
